% Shared preamble for the Mytheus research documents.
% Compiles with pdfLaTeX on Overleaf. No external assets, no custom classes.
% House style: no em-dashes and no en-dashes anywhere in the prose.

\usepackage[margin=2.1cm]{geometry}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{bm}
\usepackage{mathptmx}
\usepackage{microtype}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{fancyhdr}
\usepackage{xcolor}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,shapes.geometric,fit,backgrounds,calc}
\usepackage[colorlinks=true,linkcolor=linknavy,urlcolor=linknavy,citecolor=linknavy]{hyperref}

\definecolor{linknavy}{HTML}{1A4E8A}
\definecolor{rulegrey}{HTML}{888888}
\definecolor{boxgrey}{HTML}{F2F2F2}
\definecolor{accent}{HTML}{1A4E8A}

% Paper entries are \section level, one per page.
\titleformat{\section}
  {\normalfont\large\bfseries\color{accent}}{\thesection.}{0.6em}{}
\titlespacing*{\section}{0pt}{0.9\baselineskip}{0.35\baselineskip}

\titleformat{\subsection}
  {\normalfont\normalsize\bfseries}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{0.65\baselineskip}{0.15\baselineskip}

\setlength{\parskip}{0.3\baselineskip}
\setlength{\parindent}{0pt}
% Let TeX stretch a little harder before it overflows the margin. Long
% unbreakable runs (paper titles, identifiers) otherwise cause overfull boxes.
\setlength{\emergencystretch}{2.5em}
\setlist[itemize]{leftmargin=1.2em,itemsep=0.15em,topsep=0.2em}
\setlist[enumerate]{leftmargin=1.4em,itemsep=0.15em,topsep=0.2em}

% Each document sets its own running head with \setdochead{...}.
\newcommand{\dochead}{Mytheus}
\newcommand{\setdochead}[1]{\renewcommand{\dochead}{#1}}

\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\footnotesize\color{rulegrey}\dochead}
\fancyhead[R]{\footnotesize\color{rulegrey}\thepage}
\renewcommand{\headrulewidth}{0.4pt}

% \paperhead{title}{authors}{venue and year}{url}{verification note}
\newcommand{\paperhead}[5]{%
  \vspace{0.15em}
  {\small\textbf{Authors:} #2\par}
  {\small\textbf{Venue:} #3\par}
  {\small\textbf{Link:} \url{#4}\par}
  {\footnotesize\color{rulegrey}\textbf{Verified:} #5\par}
  \vspace{0.25em}
  {\color{rulegrey}\hrule}
  \vspace{0.5em}
}

% Grey callout used for the "what this means for Mytheus" block.
\newsavebox{\takeawaybox}
\newenvironment{takeaway}
  {\par\vspace{0.4em}%
   \begin{lrbox}{\takeawaybox}%
   \begin{minipage}{\dimexpr\linewidth-14pt\relax}\small\vspace{2pt}}
  {\vspace{2pt}\end{minipage}\end{lrbox}%
   \noindent\colorbox{boxgrey}{\usebox{\takeawaybox}}\par\vspace{0.4em}}
